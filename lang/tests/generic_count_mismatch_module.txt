funky main() {
	print_bool(either~Either~Second { value <- 21 }.is_second());
}

pub mod either[U] {
	pub sum Either[T] {
		First {
			value <- T,
		},
		Second {
			value <- U,
		},
	}
}

teach[T] either~Either[T] {
	funky is_second(self <- Self) -> bool {
		match self {
			Self~First { .. } => true,
			else => false,
		}
	}
}
