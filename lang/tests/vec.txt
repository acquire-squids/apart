funky main() {
	let
		v1 = Vec~new(),
	in {
		v1.push(0);
		v1.push(1);
		v1.push(2);

		print_string(vec_to_string(v1));

		print_string("POP!");

		match v1.pop() {
			Option~Some { value } => {
				print_i8(value);
			}
			else => {
				panic("the vec is not empty");
			}
		}

		print_string(vec_to_string(v1));

		print_string("EXTEND!");

		v1.extend(v1.slice(0, v1.length()).expect("the vec should have two values right now"));

		print_string(vec_to_string(v1));

		print_string("TRUNCATE!");

		v1.truncate(0);

		print_string(vec_to_string(v1));

		print_string("POP!");

		if v1.pop().is_some() {
			panic("truncating the vec to length 0 left some values in it!");
		}
	};
}
