funky main() {
	let printer = double_wrap(print_unit) in {
		double_unwrap(printer)(true);
	};
}

product Wrapper[T] {
	value <- T,
}

product OtherWrapper[O] {
	value <- O,
}

funky double_wrap[W](value <- W) -> OtherWrapper[Wrapper[W]] {
	OtherWrapper {
		value <- Wrapper {
			value <- value,
		},
	}
}

funky double_unwrap[U](value <- OtherWrapper[Wrapper[U]]) -> U {
	value.value.value
}
