funky main() {
	let
		greeting = "Hello, world!",
		needle = "world",
		name = "Bingus",
		truncate_at = 0,
	in {
		while greeting.length() > truncate_at {
			match greeting
				.slice(
					greeting.floor_char_boundary(truncate_at),
					greeting.ceil_char_boundary(
						truncate_at + needle.length()
					)
			) {
				Option~Some { value, } => {
					if value == needle {
						greeting.truncate(
							greeting.floor_char_boundary(truncate_at),
						);

						greeting.extend(name);

						greeting.extend("!");

						print_string(greeting);

						return {};
					}
				}
				else => {}
			}

			truncate_at = greeting.ceil_char_boundary(truncate_at + 1);
		}
	};

	print_string("Who are you, again?");
}
