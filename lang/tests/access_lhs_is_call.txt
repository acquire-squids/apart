funky main() {
	let printer = double_wrap(print_bool) in {
		unwrap(printer).value(true);
	};
}

product Wrapper[T] {
	value <- T,
}

product OtherWrapper[O] {
	value <- O,
}

funky double_wrap[W](value <- W) -> OtherWrapper[Wrapper[W]] {
	OtherWrapper {
		value <- Wrapper {
			value <- value,
		},
	}
}

funky unwrap[U](value <- OtherWrapper[U]) -> U {
	value.value
}
