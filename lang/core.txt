pub alias Option[T] = option~Option[T];
pub alias Result[T, E] = result~Result[T, E];

native funky print_u8 = funky(u8) -> unit;
native funky print_i8 = funky(i8) -> unit;

native funky print_i16 = funky(i16) -> unit;

pub mod string {
	teach String {
		pub native funky extend = funky(Self, Self) -> unit;

		pub native funky slice = funky(Self, u64, u64) -> root~Option[Self];

		pub native funky length = funky(Self) -> u64;

		pub native funky truncate = funky(Self, u64) -> unit;

		pub native funky floor_char_boundary = funky(Self, u64) -> u64;

		pub native funky ceil_char_boundary = funky(Self, u64) -> u64;
	}
}

pub mod option {
	pub sum Option[T] {
		Some {
			pub value <- T,
		},
		None {},
	}

	teach[T] Option[T] {
		pub funky is_some(self <- Self) -> bool {
			match self {
				Self~Some { .. } => true,
				else => false,
			}
		}

		pub funky is_none(self <- Self) -> bool {
			match self {
				Self~None {} => true,
				else => false,
			}
		}
	}
}

pub mod result {
	pub sum Result[T, E] {
		Ok {
			pub value <- T,
		},
		Err {
			pub error <- E,
		},
	}

	teach[T, E] Result[T, E] {
		pub funky is_ok(self <- Self) -> bool {
			match self {
				Self~Ok { .. } => true,
				else => false,
			}
		}

		pub funky is_err(self <- Self) -> bool {
			match self {
				Self~Err { ..} => true,
				else => false,
			}
		}
	}
}
