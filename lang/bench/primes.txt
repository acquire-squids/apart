funky main() {
	let
		start_time = clock(),
		count = 0,
		at = 2,
	in {
		while at < 10_000 {
			if is_prime(at) {
				count = count + 1;
			}

			at = at + 1;
		}

		print_u64(count);

		let
			end_time = clock() - start_time,
		in {
			print_f64(end_time / 1000.0);
		};
	};
}

funky is_prime(n <- u64) -> bool {
	if n < 2 || (n != 2 && n % 2 == 0) {
		return false;
	}

	let
		i = 3,
	in {
		while i < n {
			if n % i == 0 {
				return false;
			}

			i = i + 2;
		}
	};

	true
}
