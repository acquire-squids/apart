funky main() {
	let
		start_time = clock(),
		zoo = Zoo~new(),
		total = 0,
	in {
		while total < 100000000 {
			total = total + zoo.ant()
				+ zoo.banana()
				+ zoo.tuna()
				+ zoo.hay()
				+ zoo.grass()
				+ zoo.mouse();
		}

		print_i64(total);

		let
			end_time = clock() - start_time,
		in {
			"2.694 seconds";
			print_f64(end_time / 1000.0);
		};
	};
}

product Zoo {
	aardvark <- i64,
	baboon <- i64,
	cat <- i64,
	donkey <- i64,
	elephant <- i64,
	fox <- i64,
}

teach Zoo {
	funky new() -> Self {
		Zoo {
			aardvark <- 1,
			baboon <- 1,
			cat <- 1,
			donkey <- 1,
			elephant <- 1,
			fox <- 1,
		}
	}

	funky ant(self <- Self) -> i64 {
		self.aardvark
	}

	funky banana(self <- Self) -> i64 {
		self.baboon
	}

	funky tuna(self <- Self) -> i64 {
		self.cat
	}

	funky hay(self <- Self) -> i64 {
		self.donkey
	}

	funky grass(self <- Self) -> i64 {
		self.elephant
	}

	funky mouse(self <- Self) -> i64 {
		self.fox
	}
}
